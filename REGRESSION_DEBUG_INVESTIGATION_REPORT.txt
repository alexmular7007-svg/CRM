    HttpEntity<Map<String, Object>> entity = new HttpEntity<>(body, headers);

        var response = restTemplate.exchange(
            "https://api.brevo.com/v3/smtp/email",
            HttpMethod.POST,
            entity,
            String.class
        );

    } catch (HttpStatusCodeException ex) {
        log.error("STATUS = {}", ex.getStatusCode());
