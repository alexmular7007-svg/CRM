ROOT CAUSE #1: IMAGE Render Never Executes
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Location: MessageArea.jsx line 176
Issue: Checks msg.attachmentUrl (null) instead of msg.attachmentId (exists)
Result: Image render condition always FALSE
Impact: Image attachments never displayed

Before:
  {msg.messageType === 'IMAGE' && msg.attachmentUrl && (
      render image...
  )}

After:
  {msg.messageType === 'IMAGE' && msg.attachmentId && (
      render image...
  )}


ROOT CAUSE #2: FILE Render Never Executes
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Location: MessageArea.jsx line 188
Issue: Same as ROOT CAUSE #1
Result: File render condition always FALSE
Impact: File download button never displayed

Before:
  {msg.messageType === 'FILE' && msg.attachmentUrl && (
      render download button...
  )}

After:
  {msg.messageType === 'FILE' && msg.attachmentId && (
      render download button...
  )}


ROOT CAUSE #3: Download Handler Uses Wrong ID
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Location: MessageArea.jsx line 74
Issue: Passes msg.id (ChatMessage ID) instead of msg.attachmentId (Attachment ID)
Result: GET /api/attachments/60/url returns 404 (correct ID is 41)
Impact: Even if button showed, download would fail

Before:
  const signedUrl = await attachmentService.getDownloadUrl(msg.id)
  // msg.id = 60 (ChatMessage ID)
  // Endpoint: GET /api/attachments/60/url
  // Result: NOT FOUND

After:
  const signedUrl = await attachmentService.getDownloadUrl(msg.attachmentId)
  // msg.attachmentId = 41 (Attachment ID)
  // Endpoint: GET /api/attachments/41/url
  // Result: FOUND → Signed URL generated
